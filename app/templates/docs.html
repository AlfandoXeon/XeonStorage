Python (requests): 
import requests

API_KEY = "XST_your_api_key_here"
BASE_URL = "{{ request.base_url }}"
headers = {"Authorization": f"Bearer {API_KEY}"}

# 1. Upload File (MP4, Image, Doc)
with open("sample.mp4", "rb") as f:
    r = requests.post(f"{BASE_URL}api/v1/files", headers=headers, files={"file": f})
    data = r.json()
    print("Direct URL:", data["data"]["url"])
    file_id = data["data"]["id"]

# 2. Get Metadata
info = requests.get(f"{BASE_URL}api/v1/files/{file_id}", headers=headers).json()
print("File Info:", info)

# 3. List My Files
my_files = requests.get(f"{BASE_URL}api/v1/me/files", headers=headers).json()
print("Total Files:", len(my_files["data"]))
JavaScript / Node.js (Fetch & FormData): 
const fs = require('fs');
const FormData = require('form-data');
const axios = require('axios');

const API_KEY = 'XST_your_api_key_here';
const BASE_URL = '{{ request.base_url }}';

async function uploadFile() {
  const form = new FormData();
  form.append('file', fs.createReadStream('photo.jpg'));

  const res = await axios.post(`${BASE_URL}api/v1/files`, form, {
    headers: {
      'Authorization': `Bearer ${API_KEY}`,
      ...form.getHeaders()
    }
  });

  console.log('Upload Result:', res.data);
}

uploadFile();

{{ t('errors_title') }}

Status Code Description Details
200 OK Request processed successfully. Standard success.
206 Partial Content Media range stream seeking. Normal for video/audio players.
304 Not Modified ETag cache matched. 0 latency, saves 100% bandwidth.
401 Unauthorized Invalid or missing API Key. Check Authorization Bearer token header.
404 Not Found File or endpoint not found. Check requested file_id.
413 Payload Too Large File size exceeds 1.5 GB limit. Compress file under 1.5 GB.
500 Internal Error Server / Telegram API communication error. System error handler.
